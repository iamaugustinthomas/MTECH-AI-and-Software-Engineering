\section{Example for every type of set from AI and ML Domain}
\begin{frame}{Example for every type of set from AI and ML Domain }
    \begin{itemize}
        \item \textbf{Universal Set ($\mathcal{U}$):} The set of all elements under consideration, meaning any set $A$ is a subset of it ($A \subseteq \mathcal{U}$). 
        \\ \textit{ML Example:} All possible emails that a spam filter could ever receive in the real world.
        
        \vspace{0.5cm}
        \item \textbf{Finite Set:} A set with a countable, limited number of elements, where its size (cardinality) is $|S| = n$ for some integer $n$. 
        \\ \textit{ML Example:} The target labels in a sentiment analysis model: $\{\text{Positive}, \text{Neutral}, \text{Negative}\}$.
    \end{itemize}
\end{frame}

\begin{frame}{Example for every type of set from AI and ML Domain }
    \begin{itemize}
        \item \textbf{Infinite Set:} A set with an endless number of elements, where $|S| = \infty$. 
        \\ \textit{ML Example:} All the possible decimal values a neural network weight can take between $0$ and $1$, represented mathematically as the continuous interval $[0, 1]$.
        
        \vspace{0.5cm}
        \item \textbf{Empty Set ($\emptyset$):} A set containing zero elements, meaning $|\emptyset| = 0$. 
        \\ \textit{ML Example:} The overlap (intersection) between a properly split Training set ($A$) and Testing set ($B$). To avoid leakage, they share no elements: $A \cap B = \emptyset$.
    \end{itemize}
\end{frame}

\begin{frame}{Example for every type of set from AI and ML Domain }
    \begin{itemize}
        \item \textbf{Singleton Set:} A set containing exactly one element, meaning $|S| = 1$, or $S = \{x\}$.
        \\ \textit{ML Example:} A batch of data containing exactly one sample, used when a model is making a single real-time prediction in production.
        
        \vspace{0.5cm}
        \item \textbf{Subset ($A \subseteq B$):} Every element in $A$ is also contained in $B$, defined logically as $\forall x (x \in A \implies x \in B)$.
        \\ \textit{ML Example:} The Validation dataset is a subset of the total collected historical data.
    \end{itemize}
\end{frame}

\begin{frame}{Example for every type of set from AI and ML Domain }
    \begin{itemize}
        \item \textbf{Disjoint Sets:} Two sets that share no common elements whatsoever, defined as $A \cap B = \emptyset$.
        \\ \textit{ML Example:} The set of images labeled "Cat" ($A$) and the set labeled "Car" ($B$), assuming an image can only be exactly one of these.
    \end{itemize}
\end{frame}

\begin{frame}{Union of Sets}
The Union of Sets A and B denoted by A $\cup$ B, is the set of distinct elements that belong to set A or set B, or both.\\
The operation can be represented as:\\
\begin{center}
    $A\cup B=\{x|x\in A\ or\ x\in B\}$
\end{center}
 The union includes every element that appears in either of the two sets, without any repetition.

\end{frame}

\section{General form of Inclusion-Exclusion Principle}
\begin{frame}{General form of Inclusion-Exclusion Principle}

For finite sets $A_1, A_2, A_3, \dots, A_n$, the total number of elements in their union is:
\vspace{0.5em}

\[ 
\begin{aligned}
\left\lvert \bigcup_{i=1}^{n} A_i \right\rvert &= \sum_{i=1}^{n} |A_i| \\
&\quad - \sum_{1 \le i < j \le n} |A_i \cap A_j| \\
&\quad + \sum_{1 \le i < j < k \le n} |A_i \cap A_j \cap A_k| \\
&\quad - \cdots \\
&\quad + (-1)^{n-1} |A_1 \cap A_2 \cap \cdots \cap A_n|
\end{aligned}
\] 

\end{frame}
\begin{frame}{General form of Inclusion-Exclusion Principle}

For finite sets $A_1, A_2, A_3, \dots, A_n$, the total number of elements in their union is:
\vspace{1em}

\[ 
\left\lvert \bigcup_{i=1}^{n} A_i \right\rvert = \sum_{k=1}^{n} (-1)^{k-1} \left( \sum_{1 \le i_1 < i_2 < \dots < i_k \le n} |A_{i_1} \cap A_{i_2} \cap \dots \cap A_{i_k}| \right)
\] 

\end{frame}

\section{Poset}
\begin{frame}{What is a Poset?}
A partially ordered set is a a set where we have defined a specific rule fro how elements in a set relate to each other.\\
\vspace{1 ex}
For a set to be a poset, the following three conditions must be met 
\begin{itemize}
    \item Reflexive: Every item relates to itself.
    \item Antisymmetric: If item $\mathbf{A}$ relates to item $\mathbf{B}$ and item $\mathbf{B}$ relates back to item $\mathbf{A}$, then they must actaually be the exact same item.
    \item Transitive If $\mathbf{A}$ relates to $\mathbf{B}$ and $\mathbf{B}$  relates to $\mathbf{C}$ , then $\mathbf{A}$ automatically relates to $\mathbf{C}$ 
\end{itemize}

    
\end{frame}
